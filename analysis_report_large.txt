        s.row_number,
        s.seat_number,
        CASE
            WHEN t.id IS NULL THEN 'свободно'
            ELSE 'занято'
        END AS status
    FROM seat s
    LEFT JOIN ticket t ON s.id = t.seat_id AND t.showtime_id = 1

======================================================================

        MIN(st.base_price + COALESCE(sty.price_modifier, 0)) AS min_price,
